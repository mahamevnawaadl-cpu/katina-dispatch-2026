      <p><b>Remaining to Dispatch:</b> <span id="remainingToDispatch">-</span></p>

      <p><b>Remaining Pirikara:</b> <span id="remainingPirikara">-</span></p>

    </div>


    <input type="text" id="dispatcherName" placeholder="Enter your name (Dispatcher)">

    

    <button id="scanBtn">📷 Open Live Scanner</button>


    <div id="reader-container">

      <div id="reader" style="width: 100%;"></div>

      <button id="closeScanBtn" style="background-color: #dc3545; margin: 10px; width: calc(100% - 20px);">Close Camera</button>

    <button id="searchBtn">Search</button>


    <div id="results"></div>

    <div id="message"></div>


    <script>

      const API_URL = "https://script.google.com/macros/s/AKfycbwIuTBrETeeX6qVuLT-naCqkAEXJJRNLfW1vGT9ttFzUI9zivWjv3bM9QGXz-EmFBVM/exec"; // <-- Paste your URL here


      const searchBox = document.getElementById("searchBox");

      const dispatcherInput = document.getElementById("dispatcherName");

      const resultsDiv = document.getElementById("results");

      const messageBox = document.getElementById("message");

      const scanBtn = document.getElementById("scanBtn");

      const readerContainer = document.getElementById("reader-container");

      const closeScanBtn = document.getElementById("closeScanBtn");


      let html5QrCode = null;


      window.onload = () => {

        const savedDispatcher = localStorage.getItem("dispatcherName");

        if (savedDispatcher) dispatcherInput.value = savedDispatcher;

        fetchSummary();

      };


      dispatcherInput.addEventListener("input", () => {

        localStorage.setItem("dispatcherName", dispatcherInput.value.trim());

      });


      function fetchSummary() {

        fetch(API_URL + "?action=summary")

          .then(res => res.json())

          .then(data => {

            document.getElementById("totalQuantity").innerText = data.totalQuantity || "-";

            document.getElementById("dispatched").innerText = data.dispatched || "-";

            document.getElementById("remainingToDispatch").innerText = data.remainingToDispatch || "-";

            document.getElementById("remainingPirikara").innerText = data.remainingPirikara || "-";

          })

          .catch(err => console.error("Summary error:", err));

      }


      scanBtn.addEventListener("click", () => {

        const dispatcher = dispatcherInput.value.trim();

        if (!dispatcher) { alert("Please enter your name before scanning."); return; }


        readerContainer.style.display = "block";

        scanBtn.style.display = "none";


        if (!html5QrCode) {

          html5QrCode = new Html5Qrcode("reader");

        }


        html5QrCode.start(

          { facingMode: "environment" }, 

          { fps: 15, qrbox: { width: 250, height: 250 } }, 

          (decodedText) => {

            stopCamera();

            searchBox.value = decodedText.trim();

            doSearch();

          },

          (err) => {}

        ).catch(err => {

          alert("Camera error: " + err);

          stopCamera();

        });

      });


      closeScanBtn.addEventListener("click", stopCamera);


      function stopCamera() {

        if (html5QrCode) {

          html5QrCode.stop().then(() => {

            readerContainer.style.display = "none";

            scanBtn.style.display = "block";

          }).catch(() => {

            readerContainer.style.display = "none";

            scanBtn.style.display = "block";

          });

        } else {

          readerContainer.style.display = "none";

          scanBtn.style.display = "block";

        }

      }


      document.getElementById("searchBtn").addEventListener("click", doSearch);

      searchBox.addEventListener("keypress", (e) => {

        if (e.key === "Enter") { e.preventDefault(); doSearch(); }

      });


      function doSearch() {

        const query = searchBox.value.trim();

        const dispatcher = dispatcherInput.value.trim();

        if (!dispatcher) { alert("Please enter your name before searching."); return; }

        if (!query) { alert("Please enter a search term."); return; }


        resultsDiv.innerHTML = "<p><i>Searching...</i></p>";

        fetchSummary();


        fetch(`${API_URL}?action=search&query=${encodeURIComponent(query)}`)

          .then(res => res.json())

          .then(results => displayResults(results))

          .catch(err => { resultsDiv.innerHTML = "<p style='color:red;'>Search error.</p>"; });


        searchBox.value = "";

      }


      function displayResults(results) {

        if (!results || results.length === 0) {

          resultsDiv.innerHTML = "<p style='color:red;'>No records found.</p>";

          return;

        }


        resultsDiv.innerHTML = "";

        results.forEach(r => {

          const isDispatched = r["Donation Status"].toLowerCase() === "dispatched";

          const card = document.createElement("div");

          card.className = "card " + (isDispatched ? "dispatched" : "pending");


          card.innerHTML = `

            <h3>Donation #${r["Donation Number"]}</h3>

            <p><b>Name:</b> <span style="font-size:1.6em;font-weight:bold;">${r["First Name"]} ${r["Last Name"]}</span></p>

            <p><b>Quantity:</b> <span style="font-size:1.6em;font-weight:bold;">${r["Quantity"]}</span></p>

            <p><b>Status:</b> ${r["Donation Status"]}</p>

            <p><b>Dispatched By:</b> ${r["Dispatched By"] || "-"}</p>

            <p><b>Dispatched Time:</b> ${r["Dispatched Time"] || "-"}</p>

          `;


          const btn = document.createElement("button");

          btn.className = "dispatch-btn";

          btn.innerText = isDispatched ? "Already Dispatched" : "Dispatch";

          btn.disabled = isDispatched;


          btn.addEventListener("click", () => {

            const dispatcher = dispatcherInput.value.trim();

            if (!dispatcher) { alert("Dispatcher name missing."); return; }


            btn.disabled = true;

            fetch(`${API_URL}?action=dispatch&row=${r.row}&dispatcher=${encodeURIComponent(dispatcher)}`)

              .then(res => res.json())

              .then(data => {

                if (data.status === "SUCCESS") {

                  showMessage("✅ Successfully Dispatched!");

                  r["Donation Status"] = "Dispatched";

                  r["Dispatched By"] = dispatcher;

                  r["Dispatched Time"] = new Date().toLocaleString();

                  card.classList.remove("pending");

                  card.classList.add("dispatched");

                  btn.innerText = "Already Dispatched";

                  fetchSummary();

                } else if (data.status === "ALREADY_DISPATCHED") {

                  showMessage("⚠️ Already Dispatched!");

                  card.classList.remove("pending");

                  card.classList.add("dispatched");

                  btn.innerText = "Already Dispatched";

                  fetchSummary();

                } else {

                  alert("Error updating dispatch.");

                  btn.disabled = false;

                }

              });

          });


          card.appendChild(btn);

          resultsDiv.appendChild(card);

        });

      }


      function showMessage(msg) {

        messageBox.innerText = msg;

        messageBox.style.display = "block";

        setTimeout(() => { messageBox.style.display = "none"; }, 1500);

      }
